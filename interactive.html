Wireless Gate Monitor

ESP32-C3 Sender + CYD Receiver (UDP Edition)

Wireless Dual-Board UDP

Welcome to the updated build guide for the Wireless Gate Monitor. This system decouples the sensor from the screen using two boards communicating over Wi-Fi (UDP). The small ESP32-C3 sits at the gate (Sender), broadcasting status to the ESP32-CYD display (Receiver) located comfortably inside your home.

1. The Dual-Board Kit

Hover over the components to see their role in the wireless ecosystem.

SENDER
📡

ESP32-C3 SuperMini

Tiny Low-Power Board

Located at the Gate. Reads the magnetic switch and broadcasts UDP packets ("OPEN" or "CLOSED") to the house.
RECEIVER
🖥️

The "CYD" Display

ESP32-3248S035

Located in the House. Connects to Wi-Fi, displays weather, and listens for status updates from the Gate. No wiring required!
SENSOR
🧲

Magnetic Switch

Wired to Sender

Connects physically to the ESP32-C3 at the gate. Simple 2-wire connection (GND and GPIO).

Sender_ESP32C3.ino

Board: ESP32C3 Dev Module 
#include <WiFi.h>
#include <WiFiUdp.h>

// --- CONFIGURATION ---
const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";
const int UDP_PORT = 4210;

// --- PINS (ESP32-C3 SuperMini) ---
#define SWITCH_PIN 3  // Connect N.C. wire here
#define LED_PIN 8     // Onboard LED (Active Low)

WiFiUDP udp;
IPAddress broadcastIP(255, 255, 255, 255);

bool lastState = false;
unsigned long lastSentTime = 0;

void setup() {
  Serial.begin(115200);
  pinMode(SWITCH_PIN, INPUT_PULLUP);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, HIGH); // Off

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(200);
    digitalWrite(LED_PIN, LOW); // Blink
    delay(200);
    digitalWrite(LED_PIN, HIGH);
  }
  digitalWrite(LED_PIN, LOW); // LED On = Connected
}

void loop() {
  bool isOpen = (digitalRead(SWITCH_PIN) == HIGH);

  if (isOpen != lastState || millis() - lastSentTime > 1000) {
    sendPacket(isOpen);
    lastState = isOpen;
    lastSentTime = millis();
  }
  delay(100);
}

void sendPacket(bool isOpen) {
  udp.beginPacket(broadcastIP, UDP_PORT);
  if (isOpen) udp.print("OPEN");
  else udp.print("CLOSED");
  udp.endPacket();
  
  digitalWrite(LED_PIN, HIGH); // Blink
  delay(50);
  digitalWrite(LED_PIN, LOW);
}

Receiver_CYD.ino

Board: ESP32 Dev Module 
#include <WiFi.h>
#include <WiFiUdp.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <TFT_eSPI.h> 

// --- CONFIGURATION ---
const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";
const int UDP_PORT = 4210;

const float LATITUDE = 39.24;
const float LONGITUDE = -94.42;

#define RED_LED 22
#define GREEN_LED 16
#define BLUE_LED 17

TFT_eSPI tft = TFT_eSPI(); 
WiFiUDP udp;
char packetBuffer[255];
String weatherStr = "Waiting...";
unsigned long lastWeatherTime = 0;
unsigned long lastPacketTime = 0;
int currentState = 2; // 2=Lost Signal
int lastDrawnState = -1;

void setup() {
  Serial.begin(115200);
  pinMode(RED_LED, OUTPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(BLUE_LED, OUTPUT);
  
  tft.init();
  tft.setRotation(1);
  tft.fillScreen(TFT_BLACK);
  tft.setTextColor(TFT_WHITE, TFT_BLACK);
  tft.setTextSize(2);
  tft.println("Connecting WiFi...");

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  
  tft.println("Listening for Gate...");
  udp.begin(UDP_PORT);
  fetchWeather();
}

void loop() {
  int packetSize = udp.parsePacket();
  if (packetSize) {
    int len = udp.read(packetBuffer, 255);
    if (len > 0) packetBuffer[len] = 0;
    String msg = String(packetBuffer);
    lastPacketTime = millis();
    
    if (msg == "OPEN") currentState = 1;
    else if (msg == "CLOSED") currentState = 0;
  }

  if (millis() - lastPacketTime > 5000) currentState = 2;

  if (currentState != lastDrawnState) {
    drawScreen(currentState);
    lastDrawnState = currentState;
  }

  if (millis() - lastWeatherTime > 900000) {
    fetchWeather();
    drawScreen(currentState); 
  }
  delay(100);
}

void drawScreen(int state) {
  if (state == 1) { 
    tft.fillScreen(TFT_RED);
    tft.setTextColor(TFT_WHITE, TFT_RED);
    tft.setTextDatum(MC_DATUM);
    tft.drawString("GATE OPEN!", 160, 120, 6);
    digitalWrite(RED_LED, LOW); digitalWrite(GREEN_LED, HIGH); digitalWrite(BLUE_LED, HIGH);
  } else if (state == 0) {
    tft.fillScreen(TFT_GREEN);
    tft.setTextColor(TFT_BLACK, TFT_GREEN);
    tft.setTextDatum(MC_DATUM);
    tft.drawString("GATE CLOSED", 160, 100, 6);
    tft.setTextFont(4);
    tft.drawString(weatherStr, 160, 180);
    digitalWrite(RED_LED, HIGH); digitalWrite(GREEN_LED, LOW); digitalWrite(BLUE_LED, HIGH);
  } else {
    tft.fillScreen(TFT_ORANGE);
    tft.setTextColor(TFT_WHITE, TFT_ORANGE);
    tft.setTextDatum(MC_DATUM);
    tft.drawString("NO SIGNAL", 160, 100, 6);
    digitalWrite(RED_LED, HIGH); digitalWrite(GREEN_LED, HIGH); digitalWrite(BLUE_LED, LOW);
  }
}

void fetchWeather() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin("https://api.open-meteo.com/v1/forecast?latitude=" + String(LATITUDE) + "&longitude=" + String(LONGITUDE) + "¤t_weather=true&temperature_unit=fahrenheit");
    int httpCode = http.GET();
    if (httpCode > 0) {
      String payload = http.getString();
      DynamicJsonDocument doc(1024);
      deserializeJson(doc, payload);
      float temp = doc["current_weather"]["temperature"];
      weatherStr = String(temp, 1) + "F";
    }
    http.end();
    lastWeatherTime = millis();
  }
}
